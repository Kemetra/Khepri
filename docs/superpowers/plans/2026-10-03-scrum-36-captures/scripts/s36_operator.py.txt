"""SCRUM-36 operator setup, run inside khepri-staging-web as the application role.

The approved local identity handoff the M4 run used: production verbs only. Accounts,
organizations, sessions and the active-organization switch are created through the RCA services;
beta invitations through `InvitationService.issue_invitation`, the operator action RRA-001 keeps
off HTTP. Nothing is written to a table directly. Prints one JSON object on stdout.
"""

from __future__ import annotations

import json
import os
import secrets
import sys
from datetime import timedelta

from khepri.rca.accounts import AccountService
from khepri.rca.organizations import OrganizationService
from khepri.rca.persistence import SqlAccountStore, SqlOrganizationStore
from khepri.rca.session_persistence import SqlSessionStore as SqlRcaSessionStore
from khepri.rca.session_service import SessionService
from khepri.rca.switching import OrganizationSwitcher
from khepri.runtime.config import RuntimeSettings
from khepri.runtime.db_roles import DatabaseRole
from khepri.runtime.wiring import KHEPRI_SESSION_LIFETIME, build_stack


def member(stack, accounts, organizations, sessions, switcher, tag: str) -> dict:
    now = stack.clock()
    run = os.environ.get("S36_RUN", "1")
    account = accounts.create_account(f"s36-{tag}-{run}@example.test", secrets.token_urlsafe(18))
    organization = organizations.create_organization(f"S36 {tag.upper()} {run}", account.account_id, now=now)
    token = sessions.create(account.account_id, now=now)
    switcher.switch(token, organization.organization_id, now=now)
    return {"organization_id": organization.organization_id, "session_token": token}


def main() -> None:
    # Stdout is this script's only way out of the container, so it carries tokens. It must be
    # redirected to an operator-only file; a terminal would display them.
    if sys.stdout.isatty():
        raise SystemExit("redirect stdout to an operator-only file: it carries session tokens")
    stack = build_stack(RuntimeSettings.from_environment(role=DatabaseRole.APPLICATION))
    account_store = SqlAccountStore(stack.factory)
    organization_store = SqlOrganizationStore(stack.factory)
    accounts = AccountService(account_store)
    organizations = OrganizationService(organization_store)
    sessions = SessionService(SqlRcaSessionStore(stack.factory), lifetime=KHEPRI_SESSION_LIFETIME)
    switcher = OrganizationSwitcher(sessions, organization_store)
    out = {
        "a": member(stack, accounts, organizations, sessions, switcher, "a"),
        "b": member(stack, accounts, organizations, sessions, switcher, "b"),
        "invitations": [
            stack.services.invitations.issue_invitation(
                expires_at=stack.clock() + timedelta(days=7)
            )
            for _ in range(6)
        ],
    }
    print(json.dumps(out))


main()
